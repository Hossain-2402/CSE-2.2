.MODEL SMALL
.STACK 100H
.DATA

    PATTERNS DB 7EH, 09H, 09H, 09H, 7EH     
     
     
     
     
.CODE
MAIN PROC
    
    
    MOV AX, @DATA
    MOV DS, AX
    
    MOV CX, 5
    

    LEA SI, PATTERNS
    MOV DX, 2000H
    

WRITE_LOOP:
    
    MOV AL, [SI]
    OUT DX, AL
    INC SI
    INC DX
    
    LOOP WRITE_LOOP
    
    HLT
    
MAIN ENDP
END MAIN




.MODEL SMALL
.STACK 100H
.DATA

    N DB ?
    M DB ? 
    
    ONE DB 1000100B, 1000010B, 1111111B, 1000000B, 1000000B 
    FOUR DB 0001100B, 0001010B, 1111101B, 0000000B, 0000000B 
    
    PORT DW 2000H
    
    PATTERN DB 0000000B, 0000001B, 0000011B, 0000111B, 0001111B, 0011111B, 0111111B,

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX
    
          
    MOV AH, 1
    INT 21H    
    SUB AL, 30H
    MOV N, AL


    
    XOR AX, AX
    MOV AH, 1
    INT 21H   
    SUB AL, 30H
    MOV M, AL 
    
                                                   
                
 
    L:              
         
        LEA SI, PATTERN
        XOR DX, DX
        MOV DX, PORT
 
 
        XOR BX, BX
        MOV BL, N           
        ADD SI, BX                           
        MOV AL, [SI]
        OUT DX, AL  
                     
        CALL DELAY                    
                     
        MOV AL, 0000000B
        OUT DX, AL
              

        XOR BX, BX 
        MOV BL, M 
        ADD PORT, BX
        
        CMP PORT, 2005H
        JE GO_LEFT


CNG_DIR:
    CMP M, 0
    JG GO_LEFT
    JMP GO_RIGHT
 
GO_LEFT:
    MOV BL, M
    MOV BL, -M
    MOV M, BL
    
    JMP L 
    
         
GO_RIGHT:
    MOV BX, 0
    MOV BL, -M
    MOV M, BL
    
    JMP L


MAIN ENDP       

DELAY PROC
    MOV CX, 5
    L1:
        LOOP L1  
        RET
DELAY ENDP       



END MAIN
    
END MAIN
MAIN ENDP
